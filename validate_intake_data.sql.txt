CREATE TABLE patient_intake_data (
    patient_id VARCHAR(20) PRIMARY KEY,
    age INT,
    gender VARCHAR(10),
    primary_complaint VARCHAR(255)
);

SELECT patient_id,
       age,
       gender,
       primary_complaint
FROM patient_intake_data
WHERE primary_complaint IS NULL;

SELECT patient_id,
       COUNT(*) AS record_count
FROM patient_intake_data
GROUP BY patient_id
HAVING COUNT(*) > 1;
